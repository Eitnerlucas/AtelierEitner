#### 2. Compresión Automática de Imágenes (para máxima velocidad)
```javascript
itemImageInput.addEventListener('change', (e) => {
  const file = e.target.files[0];
  if (!file) return;

  const reader = new FileReader();
  reader.onload = (event) => {
    const img = new Image();
    img.src = event.target.result;
    img.onload = () => {
      const canvas = document.createElement('canvas');
      const MAX_SIZE = 1000; // Máximo 1000px para que sea liviana
      let width = img.width;
      let height = img.height;

      if (width > height) {
        if (width > MAX_SIZE) {
          height *= MAX_SIZE / width;
          width = MAX_SIZE;
        }
      } else {
        if (height > MAX_SIZE) {
          width *= MAX_SIZE / height;
          height = MAX_SIZE;
        }
      }

      canvas.width = width;
      canvas.height = height;
      const ctx = canvas.getContext('2d');
      ctx.drawImage(img, 0, 0, width, height);

      // Convierte a JPEG comprimido
      currentBase64Image = canvas.toDataURL('image/jpeg', 0.82);
    };
  };
  reader.readAsDataURL(file);
});
```

function getAllCatalogItems() {
  let customItems = [];
  try {
    const stored = localStorage.getItem(STORAGE_KEY);
    if (stored) {
      customItems = JSON.parse(stored);
    }
  } catch (e) {
    console.warn('Error al leer productos:', e);
  }
  // Los productos nuevos aparecen PRIMERO
  return [...customItems, ...itemsCatalogo];
}

#### 2. Renderizado de Tarjetas y Manejo de Visibilidad de Imagen
```javascript
const img = card.querySelector('.card-img');
const wrapper = card.querySelector('.card-img-wrapper');

const onImageLoad = () => {
  if (wrapper) wrapper.classList.add('loaded'); // Activa la visibilidad opacity: 1
};

if (img) {
  img.onload = onImageLoad;
  img.onerror = onImageLoad;
  // Si es un formato base64 o ya cargó, lo muestra inmediatamente
  if (img.complete || img.naturalWidth > 0 || (img.src && img.src.startsWith('data:'))) {
    onImageLoad();
  }
}
```

## 3. SQL de la Base de Datos (`productos`)

Este es el script SQL ejecutado en Supabase para habilitar la tabla y el acceso público de lectura/escritura:

```sql
CREATE TABLE IF NOT EXISTS public.productos (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  src text NOT NULL,
  category text NOT NULL,
  title text NOT NULL,
  "desc" text,
  created_at timestamptz DEFAULT now()
);

-- Permite lectura y carga directa desde el panel admin sin bloqueos
ALTER TABLE public.productos DISABLE ROW LEVEL SECURITY;
```
